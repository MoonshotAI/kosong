A test-only chat provider that streams parts described by a tiny DSL.

The DSL lives in the content of the last message in history and is made of lines in the form kind: payload. Empty lines, comment lines starting with #, and markdown fences starting with ``` are ignored. Supported kinds:

  • id: sets the streamed message id.
  • usage: token usage, e.g. usage: {"input_other": 10, "output": 2} or usage: input_other=1 output=2 input_cache_read=3.
  • text: a text chunk.
  • think: a thinking chunk.
  • image_url: either a raw URL or {"url": "...", "id": "opt"}.
  • audio_url: either a raw URL or {"url": "...", "id": "opt"}.
  • video_url: either a raw URL or {"url": "...", "id": "opt"}.
  • tool_call: a JSON or key/value object. Fields: id, name (or function.name), optional arguments/function.arguments, optional extras.
  • tool_call_part: a string/JSON with arguments_part; null becomes None.

28class ScriptedEchoChatProvider:
29    """
30    A test-only chat provider that consumes a queue of echo DSL scripts per call.
31    """
32
33    name = "scripted_echo"
34
35    def __init__(self, scripts: Iterable[str], *, trace: bool = False):
36        self._scripts = deque(scripts)
37        self._turn = 0
38        self._trace = trace
39
40    @property
41    def model_name(self) -> str:
42        return "scripted_echo"
43
44    @property
45    def thinking_effort(self) -> ThinkingEffort | None:
46        return None
47
48    async def generate(
49        self,
50        system_prompt: str,
51        tools: Sequence[Tool],
52        history: Sequence[Message],
53    ) -> ScriptedEchoStreamedMessage:
54        if not self._scripts:
55            raise ChatProviderError(f"ScriptedEchoChatProvider exhausted at turn {self._turn + 1}.")
56        script_text = self._scripts.popleft()
57        if self._trace:
58            script_json = json.dumps(script_text)
59            print(f"SCRIPTED_ECHO TURN {self._turn + 1}: {script_json}")
60        self._turn += 1
61        parts, message_id, usage = parse_echo_script(script_text)
62        if not parts:
63            raise ChatProviderError("ScriptedEchoChatProvider DSL produced no streamable parts.")
64        return ScriptedEchoStreamedMessage(parts=parts, message_id=message_id, usage=usage)
65
66    def with_thinking(self, effort: ThinkingEffort) -> Self:
67        copied = copy.copy(self)
68        copied._scripts = deque(self._scripts)
69        return copied
