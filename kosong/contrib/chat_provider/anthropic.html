104# Models that accept adaptive thinking but don't expose a major.minor version
105# in their identifier (e.g. Claude Mythos Preview).
106_ADAPTIVE_MARKERS_NO_VERSION: tuple[str, ...] = ("mythos",)
107
108# Matches the Claude family's major.minor version inside a model identifier.
109# `\d{1,2}(?!\d)` prevents date suffixes like `sonnet-4-20250514` from being
110# misread as minor=20: `20` would be followed by another digit, so the
111# negative lookahead fails and the regex does not match at all.
112_FAMILY_VERSION_RE = re.compile(r"(?:opus|sonnet|haiku)[.-](\d+)[.-](\d{1,2})(?!\d)")
113
114# Adaptive thinking was introduced with Opus 4.6 / Sonnet 4.6.
115_ADAPTIVE_MIN_VERSION: tuple[int, int] = (4, 6)

118def _supports_adaptive_thinking(model: str) -> bool:
119    """Whether the given model id accepts `thinking: {type: "adaptive"}`.
120
121    Strategy: explicit marker for non-versioned models (e.g. Mythos), plus a
122    strict regex over the Claude family that extrapolates to unknown future
123    versions (>= 4.6) without code changes.
124    """
125    m = model.lower()
126    if any(marker in m for marker in _ADAPTIVE_MARKERS_NO_VERSION):
127        return True
128    if match := _FAMILY_VERSION_RE.search(m):
129        major, minor = int(match.group(1)), int(match.group(2))
130        return (major, minor) >= _ADAPTIVE_MIN_VERSION
131    return False

134def _is_opus_4_7(model: str) -> bool:
135    """Opus 4.7 specifically supports the ``xhigh`` effort level.
136
137    The docs explicitly enumerate ``xhigh`` support as "Available on Claude
138    Opus 4.7" — not "4.7 and later". We keep the check exact so that a
139    future Opus 4.8 that silently drops xhigh doesn't start returning 400.
140    Future versions fall back to the 4.6-family effort set (which still
141    covers ``max``) until this table is updated.
142    """
143    m = model.lower()
144    if match := re.search(r"opus[.-](\d+)[.-](\d{1,2})(?!\d)", m):
145        major, minor = int(match.group(1)), int(match.group(2))
146        return (major, minor) == (4, 7)
147    return False

150def _supported_efforts(model: str) -> frozenset["ThinkingEffort"]:
151    """Effort levels accepted by ``output_config.effort`` for the given model.
152
153    Per Anthropic docs:
154      - xhigh: Opus 4.7 only
155      - max:   Mythos, Opus 4.7, Opus 4.6, Sonnet 4.6 (and future adaptive models)
156      - low/medium/high: all models with effort support (including Opus 4.5+)
157    """
158    if _is_opus_4_7(model):
159        return frozenset({"low", "medium", "high", "xhigh", "max"})
160    if _supports_adaptive_thinking(model):
161        # 4.6 family / Mythos / future adaptive models: support max but not xhigh
162        return frozenset({"low", "medium", "high", "max"})
163    # Pre-4.6 models: capped at high
164    return frozenset({"low", "medium", "high"})

167def _clamp_effort(effort: "ThinkingEffort", model: str) -> "ThinkingEffort":
168    """Clamp an effort level to the highest one supported by the model.
169
170    Anything the model doesn't support falls back to ``high`` — the
171    universally available ceiling. ``off`` is passed through unchanged as
172    it represents "disable thinking" rather than an effort rank.
173    """
174    if effort == "off":
175        return effort
176    if effort in _supported_efforts(model):
177        return effort
178    return "high"

181def _supports_effort_param(model: str) -> bool:
182    """Whether the model accepts ``output_config.effort`` at all.
183
184    Per Anthropic's effort docs, the parameter is explicitly supported on
185    Claude Mythos Preview, Claude Opus 4.7, Claude Opus 4.6, Claude Sonnet
186    4.6, and Claude Opus 4.5. Adaptive-capable models all support it via the
187    adaptive pathway. For the legacy (manual thinking) pathway, only Opus
188    4.5 is explicitly listed.
189
190    We gate ``output_config`` emission on this predicate to avoid sending
191    effort to models that would reject it with a 400 (Claude 3.x, and
192    conservatively Sonnet 4 / Sonnet 4.5 / Haiku 4.5 which are not in the
193    explicit list). A false negative here means "effort not sent, no
194    regression from pre-effort behaviour"; a false positive would mean
195    "API 400 error", so we err on the side of silence.
196    """
197    if _supports_adaptive_thinking(model):
198        return True
199    # Opus 4.5 is the only legacy (non-adaptive) model that Anthropic docs
200    # explicitly confirm supports the effort parameter.
201    m = model.lower()
202    return "opus-4-5" in m or "opus-4.5" in m

212    class GenerationKwargs(TypedDict, total=False):
213        max_tokens: int | None
214        temperature: float | None
215        top_k: int | None
216        top_p: float | None
217        # e.g., {"type": "adaptive", "display": "summarized"}
218        # or   {"type": "enabled", "budget_tokens": 1024}
219        thinking: ThinkingConfigParam | None
220        # e.g., {"effort": "high"} — soft guidance that applies to all output
221        # tokens. Used in adaptive thinking requests, and in legacy requests
222        # only when the model is on Anthropic's explicit effort-supporting
223        # list (see ``_supports_effort_param``).
224        output_config: OutputConfigParam | None
225        # e.g., {"type": "auto", "disable_parallel_tool_use": True}
226        tool_choice: ToolChoiceParam | None
227
228        beta_features: list[BetaFeatures] | None
229        extra_headers: Mapping[str, str] | None

231    def __init__(
232        self,
233        *,
234        model: str,
235        api_key: str | None = None,
236        base_url: str | None = None,
237        stream: bool = True,
238        # which process should we apply on tool result
239        tool_message_conversion: ToolMessageConversion | None = None,
240        # Must provide a max_tokens. Can be overridden by .with_generation_kwargs()
241        default_max_tokens: int,
242        metadata: MetadataParam | None = None,
243        **client_kwargs: Any,
244    ):
245        self._model = model
246        self._stream = stream
247        self._client = AsyncAnthropic(api_key=api_key, base_url=base_url, **client_kwargs)
248        self._tool_message_conversion: ToolMessageConversion | None = tool_message_conversion
249        self._metadata = metadata
250        self._generation_kwargs: Anthropic.GenerationKwargs = {
251            "max_tokens": default_max_tokens,
252            "beta_features": ["interleaved-thinking-2025-05-14"],
253        }

255    @property

259    @property
260    def thinking_effort(self) -> "ThinkingEffort | None":
261        thinking_config = self._generation_kwargs.get("thinking")
262        if thinking_config is None:
263            return None
264        if thinking_config["type"] == "disabled":
265            return "off"
266        if thinking_config["type"] == "adaptive":
267            output_config = self._generation_kwargs.get("output_config") or {}
268            effort = output_config.get("effort")
269            if effort in ("low", "medium", "high", "xhigh", "max"):
270                return effort
271            return "high"
272        budget = thinking_config["budget_tokens"]
273        if budget <= 1024:
274            return "low"
275        if budget <= 4096:
276            return "medium"
277        return "high"

279    async def generate(
280        self,
281        system_prompt: str,
282        tools: Sequence[Tool],
283        history: Sequence[Message],
284    ) -> "AnthropicStreamedMessage":
285        # https://docs.claude.com/en/api/messages#body-messages
286        # Anthropic API does not support system roles, but just a system prompt.
287        system = (
288            [
289                TextBlockParam(
290                    text=system_prompt,
291                    type="text",
292                    cache_control=CacheControlEphemeralParam(type="ephemeral"),
293                )
294            ]
295            if system_prompt
296            else omit
297        )
298        messages: list[MessageParam] = []
299        for message in history:
300            converted = self._convert_message(message)
301            # Per Anthropic spec, tool_result blocks for the same assistant turn
302            # must live in a single user message. Internal Messages model one
303            # tool call per entry, so merge consecutive tool-result-only user
304            # messages here. Strict-compat backends (e.g. DeepSeek /anthropic)
305            # 400 on the split form, and the official backend silently teaches
306            # the model to stop calling tools in parallel.
307            if (
308                messages
309                and converted["role"] == "user"
310                and messages[-1]["role"] == "user"
311                and _is_tool_result_only(messages[-1]["content"])
312                and _is_tool_result_only(converted["content"])
313            ):
314                prev_content = cast(list[ContentBlockParam], messages[-1]["content"])
315                new_content = cast(list[ContentBlockParam], converted["content"])
316                messages[-1]["content"] = [*prev_content, *new_content]
317            else:
318                messages.append(converted)
319        if messages:
320            last_message = messages[-1]
321            last_content = last_message["content"]
322
323            # inject cache control in the last content.
324            # https://docs.claude.com/en/docs/build-with-claude/prompt-caching
325            if isinstance(last_content, list) and last_content:
326                content_blocks = cast(list[ContentBlockParam], last_content)
327                last_block = content_blocks[-1]
328                match last_block["type"]:
329                    case (
330                        "text"
331                        | "image"
332                        | "document"
333                        | "search_result"
334                        | "tool_use"
335                        | "tool_result"
336                        | "server_tool_use"
337                        | "web_search_tool_result"
338                    ):
339                        last_block["cache_control"] = CacheControlEphemeralParam(type="ephemeral")
340                    case "thinking" | "redacted_thinking":
341                        pass
342        generation_kwargs: dict[str, Any] = {}
343        generation_kwargs.update(self._generation_kwargs)
344        betas = generation_kwargs.pop("beta_features", [])
345        extra_headers = {
346            **{"anthropic-beta": ",".join(str(e) for e in betas)},
347            **(generation_kwargs.pop("extra_headers", {})),
348        }
349
350        tools_ = [_convert_tool(tool) for tool in tools]
351        if tools:
352            tools_[-1]["cache_control"] = CacheControlEphemeralParam(type="ephemeral")
353        try:
354            response = await self._client.messages.create(
355                model=self._model,
356                messages=messages,
357                system=system,
358                tools=tools_,
359                stream=self._stream,
360                extra_headers=extra_headers,
361                metadata=self._metadata if self._metadata is not None else omit,
362                **generation_kwargs,
363            )
364            return AnthropicStreamedMessage(response)
365        except (AnthropicError, httpx.HTTPError) as e:
366            raise _convert_error(e) from e

368    def with_thinking(self, effort: "ThinkingEffort") -> Self:
369        if effort == "off":
370            new = self.with_generation_kwargs(thinking={"type": "disabled"})
371            # Clear any stale output_config from a prior adaptive configuration.
372            new._generation_kwargs.pop("output_config", None)
373            return new
374
375        # Clamp to whatever the model actually accepts. xhigh/max fall back
376        # to high on models that don't support them; low/medium/high pass
377        # through; max passes through on 4.6-family and newer.
378        effective = _clamp_effort(effort, self._model)
379        # SDK 0.78 OutputConfigParam TypedDict lists only low/medium/high/max.
380        # `xhigh` is valid on Opus 4.7 per the API docs but not yet typed.
381        output_config: OutputConfigParam = {"effort": effective}  # type: ignore[typeddict-item]
382
383        if _supports_adaptive_thinking(self._model):
384            # Opus 4.6+ / Sonnet 4.6+ / Mythos: adaptive thinking.
385            # `display: "summarized"` is required on Opus 4.7+ (where the default
386            # flipped to "omitted") and is a no-op on 4.6. Setting it
387            # unconditionally keeps thinking content visible across versions.
388            # SDK 0.78 TypedDict doesn't model `display` yet — thus the ignore.
389            thinking_config: ThinkingConfigParam = {
390                "type": "adaptive",
391                "display": "summarized",
392            }  # type: ignore[typeddict-item]
393            new = self.with_generation_kwargs(
394                thinking=thinking_config,
395                output_config=output_config,
396            )
397            # Adaptive mode auto-enables interleaved thinking, so the beta
398            # header is redundant. Drop it if still present from construction.
399            if (
400                beta_features := new._generation_kwargs.get("beta_features")
401            ) and "interleaved-thinking-2025-05-14" in beta_features:
402                beta_features.remove("interleaved-thinking-2025-05-14")
403            return new
404
405        # Pre-4.6 models: legacy budget-based thinking. After clamping,
406        # `effective` is guaranteed to be one of low/medium/high here.
407        # Only models that Anthropic's docs explicitly list as supporting the
408        # effort parameter (e.g. Opus 4.5) get `output_config` emitted; other
409        # pre-4.6 models (Sonnet 4, Sonnet 4.5, Haiku 4.5, Claude 3.x) omit it
410        # to avoid 400 validation errors on models that don't accept it.
411        budgets: dict[str, int] = {"low": 1024, "medium": 4096, "high": 32_000}
412        kwargs: dict[str, Any] = {
413            "thinking": {"type": "enabled", "budget_tokens": budgets[effective]},
414        }
415        if _supports_effort_param(self._model):
416            kwargs["output_config"] = output_config
417        return self.with_generation_kwargs(**kwargs)

419    def with_generation_kwargs(self, **kwargs: Unpack[GenerationKwargs]) -> Self:
420        """
421        Copy the chat provider, updating the generation kwargs with the given values.

423        Returns:
424            Self: A new instance of the chat provider with updated generation kwargs.
425        """
426        new_self = copy.copy(self)
427        new_self._generation_kwargs = copy.deepcopy(self._generation_kwargs)
428        new_self._generation_kwargs.update(kwargs)
429        return new_self

443    def _convert_message(self, message: Message) -> MessageParam:
444        """Convert a single internal message into Anthropic wire format."""
445        role = message.role
446
447        if role == "system":
448            # Anthropic does not support system messages in the conversation.
449            # We map it to a special user message.
450            return MessageParam(
451                role="user",
452                content=[
453                    TextBlockParam(
454                        type="text", text=f"<system>{message.extract_text(sep='\n')}</system>"
455                    )
456                ],
457            )
458        elif role == "tool":
459            if message.tool_call_id is None:
460                raise ChatProviderError("Tool message missing `tool_call_id`.")
461            if self._tool_message_conversion == "extract_text":
462                content = message.extract_text(sep="\n")
463            else:
464                content = message.content
465            block = _tool_result_message_to_block(message.tool_call_id, content)
466            return MessageParam(role="user", content=[block])
467
468        assert role in ("user", "assistant")
469        blocks: list[ContentBlockParam] = []
470        for part in message.content:
471            if isinstance(part, TextPart):
472                blocks.append(TextBlockParam(type="text", text=part.text))
473            elif isinstance(part, ImageURLPart):
474                blocks.append(_image_url_part_to_anthropic(part))
475            elif isinstance(part, ThinkPart):
476                if part.encrypted is None:
477                    # missing signature, strip this thinking block.
478                    continue
479                else:
480                    blocks.append(
481                        ThinkingBlockParam(
482                            type="thinking", thinking=part.think, signature=part.encrypted
483                        )
484                    )
485            else:
486                continue
487        for tool_call in message.tool_calls or []:
488            if tool_call.function.arguments:
489                try:
490                    parsed_arguments = json.loads(tool_call.function.arguments, strict=False)
491                except json.JSONDecodeError as exc:  # pragma: no cover - defensive guard
492                    raise ChatProviderError("Tool call arguments must be valid JSON.") from exc
493                if not isinstance(parsed_arguments, dict):
494                    raise ChatProviderError("Tool call arguments must be a JSON object.")
495                tool_input = cast(dict[str, object], parsed_arguments)
496            else:
497                tool_input = {}
498            blocks.append(
499                ToolUseBlockParam(
500                    type="tool_use",
501                    id=tool_call.id,
502                    name=tool_call.function.name,
503                    input=tool_input,
504                )
505            )
506        return MessageParam(role=role, content=blocks)

509class AnthropicStreamedMessage:
510    def __init__(self, response: AnthropicMessage | AsyncStream[RawMessageStreamEvent]):
511        if isinstance(response, AnthropicMessage):
512            self._iter = self._convert_non_stream_response(response)
513        else:
514            self._iter = self._convert_stream_response(response)
515        self._id: str | None = None
516        self._usage = Usage(input_tokens=0, output_tokens=0)
517
518    def __aiter__(self) -> AsyncIterator[StreamedMessagePart]:
519        return self
520
521    async def __anext__(self) -> StreamedMessagePart:
522        return await self._iter.__anext__()
523
524    @property
525    def id(self) -> str | None:
526        return self._id
527
528    @property
529    def trace_id(self) -> str | None:
530        return None
531
532    @property
533    def usage(self) -> TokenUsage | None:
534        # https://docs.claude.com/en/docs/build-with-claude/prompt-caching#tracking-cache-performance
535        return TokenUsage(
536            # Note: in some Anthropic-compatible APIs, input_tokens can be None
537            input_other=self._usage.input_tokens or 0,
538            output=self._usage.output_tokens,
539            input_cache_read=self._usage.cache_read_input_tokens or 0,
540            input_cache_creation=self._usage.cache_creation_input_tokens or 0,
541        )
542
543    def _update_usage(self, delta_usage: MessageDeltaUsage) -> None:
544        if delta_usage.cache_creation_input_tokens is not None:
545            self._usage.cache_creation_input_tokens = delta_usage.cache_creation_input_tokens
546        if delta_usage.cache_read_input_tokens is not None:
547            self._usage.cache_read_input_tokens = delta_usage.cache_read_input_tokens
548        if delta_usage.input_tokens is not None:
549            self._usage.input_tokens = delta_usage.input_tokens
550        if delta_usage.output_tokens is not None:  # type: ignore
551            self._usage.output_tokens = delta_usage.output_tokens
552
553    async def _convert_non_stream_response(
554        self,
555        response: AnthropicMessage,
556    ) -> AsyncIterator[StreamedMessagePart]:
557        self._id = response.id
558        self._usage = response.usage
559        for block in response.content:
560            match block.type:
561                case "text":
562                    yield TextPart(text=block.text)
563                case "thinking":
564                    yield ThinkPart(think=block.thinking, encrypted=block.signature)
565                case "redacted_thinking":
566                    yield ThinkPart(think="", encrypted=block.data)
567                case "tool_use":
568                    yield ToolCall(
569                        id=block.id,
570                        function=ToolCall.FunctionBody(
571                            name=block.name, arguments=json.dumps(block.input)
572                        ),
573                    )
574                case _:
575                    continue
576
577    async def _convert_stream_response(
578        self,
579        manager: AsyncStream[RawMessageStreamEvent],
580    ) -> AsyncIterator[StreamedMessagePart]:
581        try:
582            async with manager as stream:
583                async for event in stream:
584                    if isinstance(event, MessageStartEvent):
585                        self._id = event.message.id
586                        # Capture initial usage from start event
587                        # (contains initial prompt/input token usage)
588                        self._usage = event.message.usage
589                    elif isinstance(event, RawContentBlockStartEvent):
590                        block = event.content_block
591                        match block.type:
592                            case "text":
593                                yield TextPart(text=block.text)
594                            case "thinking":
595                                yield ThinkPart(think=block.thinking)
596                            case "redacted_thinking":
597                                yield ThinkPart(think="", encrypted=block.data)
598                            case "tool_use":
599                                yield ToolCall(
600                                    id=block.id,
601                                    function=ToolCall.FunctionBody(name=block.name, arguments=""),
602                                )
603                            case "server_tool_use" | "web_search_tool_result":
604                                # ignore
605                                continue
606                    elif isinstance(event, RawContentBlockDeltaEvent):
607                        delta = event.delta
608                        match delta.type:
609                            case "text_delta":
610                                yield TextPart(text=delta.text)
611                            case "thinking_delta":
612                                yield ThinkPart(think=delta.thinking)
613                            case "input_json_delta":
614                                yield ToolCallPart(arguments_part=delta.partial_json)
615                            case "signature_delta":
616                                yield ThinkPart(think="", encrypted=delta.signature)
617                            case "citations_delta":
618                                # ignore
619                                continue
620                    elif isinstance(event, MessageDeltaEvent):
621                        if event.usage:
622                            self._update_usage(event.usage)
623                    elif isinstance(event, MessageStopEvent):
624                        continue
625        except (AnthropicError, httpx.HTTPError) as exc:
626            raise _convert_error(exc) from exc

637def _is_tool_result_only(content: object) -> bool:
638    """True iff ``content`` is a non-empty list of only tool_result blocks.
639
640    Guards the parallel-tool-result merge in ``generate()``: we only collapse
641    consecutive user messages when both sides carry pure tool results, never
642    when a user message mixes in text, images, or anything else.
643    """
644    if not isinstance(content, list) or not content:
645        return False
646    blocks = cast(list[ContentBlockParam], content)
647    return all(b["type"] == "tool_result" for b in blocks)

650def _tool_result_message_to_block(
651    tool_call_id: str, content: str | list[ContentPart]
652) -> ToolResultBlockParam:
653    block_content: str | list[ToolResultContent]
654    # If tool_result_process is `extract_text`, we join all text parts into one string
655    if isinstance(content, str):
656        block_content = content
657    else:
658        # Otherwise, map parts to content blocks
659        blocks: list[ToolResultContent] = []
660        for part in content:
661            if isinstance(part, TextPart):
662                if part.text:
663                    blocks.append(TextBlockParam(type="text", text=part.text))
664            elif isinstance(part, ImageURLPart):
665                blocks.append(_image_url_part_to_anthropic(part))
666            else:
667                # https://docs.claude.com/en/docs/build-with-claude/files#file-types-and-content-blocks
668                # Anthropic API supports very limited file types
669                raise ChatProviderError(
670                    f"Anthropic API does not support {type(part)} in tool result"
671                )
672        block_content = blocks
673
674    return ToolResultBlockParam(
675        type="tool_result",
676        tool_use_id=tool_call_id,
677        content=block_content,
678    )

681def _image_url_part_to_anthropic(part: ImageURLPart) -> ImageBlockParam:
682    url = part.image_url.url
683    # data:[<media-type>][;base64],<data>
684    if url.startswith("data:"):
685        res = url[5:].split(";base64,", 1)
686        if len(res) != 2:
687            raise ChatProviderError(f"Invalid data URL for image: {url}")
688        media_type, data = res
689        if media_type not in ("image/png", "image/jpeg", "image/gif", "image/webp"):
690            raise ChatProviderError(
691                f"Unsupported media type for base64 image: {media_type}, url: {url}"
692            )
693        return ImageBlockParam(
694            type="image",
695            source=Base64ImageSourceParam(
696                type="base64",
697                data=data,
698                media_type=media_type,
699            ),
700        )
701    else:
702        return ImageBlockParam(
703            type="image",
704            source=URLImageSourceParam(type="url", url=url),
705        )

708def _convert_error(error: AnthropicError | httpx.HTTPError) -> ChatProviderError:
709    # httpx errors may leak through the Anthropic SDK during streaming;
710    # delegate to the shared converter.
711    if isinstance(error, httpx.HTTPError):
712        return convert_httpx_error(error)
713    # Anthropic SDK errors — check subclasses before parents to avoid
714    # misclassification (e.g. APITimeoutError inherits APIConnectionError).
715    if isinstance(error, AnthropicAPIStatusError):
716        return APIStatusError(error.status_code, str(error))
717    if isinstance(error, AnthropicAuthenticationError):
718        return APIStatusError(getattr(error, "status_code", 401), str(error))
719    if isinstance(error, AnthropicPermissionDeniedError):
720        return APIStatusError(getattr(error, "status_code", 403), str(error))
721    if isinstance(error, AnthropicRateLimitError):
722        return APIStatusError(getattr(error, "status_code", 429), str(error))
723    if isinstance(error, AnthropicAPITimeoutError):
724        return APITimeoutError(str(error))
725    if isinstance(error, AnthropicAPIConnectionError):
726        return APIConnectionError(str(error))
727    return ChatProviderError(f"Anthropic error: {error}")

444    def _convert_message(self, message: Message) -> MessageParam:
445        """Convert a single internal message into Anthropic wire format."""
446        role = message.role
447
448        if role == "system":
449            # Anthropic does not support system messages in the conversation.
450            # We map it to a special user message.
451            return MessageParam(
452                role="user",
453                content=[
454                    TextBlockParam(
455                        type="text", text=f"<system>{message.extract_text(sep='\n')}</system>"
456                    )
457                ],
458            )
459        elif role == "tool":
460            if message.tool_call_id is None:
461                raise ChatProviderError("Tool message missing `tool_call_id`.")
462            if self._tool_message_conversion == "extract_text":
463                content = message.extract_text(sep="\n")
464            else:
465                content = message.content
466            block = _tool_result_message_to_block(message.tool_call_id, content)
467            return MessageParam(role="user", content=[block])
468
469        assert role in ("user", "assistant")
470        blocks: list[ContentBlockParam] = []
471        for part in message.content:
472            if isinstance(part, TextPart):
473                blocks.append(TextBlockParam(type="text", text=part.text))
474            elif isinstance(part, ImageURLPart):
475                blocks.append(_image_url_part_to_anthropic(part))
476            elif isinstance(part, ThinkPart):
477                if part.encrypted is None:
478                    # missing signature, strip this thinking block.
479                    continue
480                else:
481                    blocks.append(
482                        ThinkingBlockParam(
483                            type="thinking", thinking=part.think, signature=part.encrypted
484                        )
485                    )
486            else:
487                continue
488        for tool_call in message.tool_calls or []:
489            if tool_call.function.arguments:
490                try:
491                    parsed_arguments = json.loads(tool_call.function.arguments, strict=False)
492                except json.JSONDecodeError as exc:  # pragma: no cover - defensive guard
493                    raise ChatProviderError("Tool call arguments must be valid JSON.") from exc
494                if not isinstance(parsed_arguments, dict):
495                    raise ChatProviderError("Tool call arguments must be a JSON object.")
496                tool_input = cast(dict[str, object], parsed_arguments)
497            else:
498                tool_input = {}
499            blocks.append(
500                ToolUseBlockParam(
501                    type="tool_use",
502                    id=tool_call.id,
503                    name=tool_call.function.name,
504                    input=tool_input,
505                )
506            )
507        return MessageParam(role=role, content=blocks)
